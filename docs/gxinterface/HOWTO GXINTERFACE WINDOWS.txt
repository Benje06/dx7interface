## GxInterface

export MSYS2_ARG_CONV_EXCL='--prefix;--datarootdir'
./autogen.sh --target=x86_64-w64-mingw32 --host=x86_64-w64-mingw32 PKG_CONFIG_PATH=/ucrt64/lib/pkgconfig --build=$(./config/config.guess) --prefix=/ucrt64

make -j3
make install
make DESTDIR=/d/gxinterface/package install

ldd /d/gxinterface/package/ucrt64/bin/gxinterface.exe | grep /ucrt64 | awk '{print $3}' | xargs -I {} cp {} /d/gxinterface/package/ucrt64/bin/
